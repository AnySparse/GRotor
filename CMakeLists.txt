option(GROTOR_BUILD_MULTIGPU "Build NVLink/NVSHMEM last-batch parallel routing" OFF)
option(GROTOR_MULTI_GRID "Distribute last-batch grid maintenance across NVSHMEM PEs" ON)
option(GROTOR_MULTI_LEGACY_LSHAPE "Use the original full-matrix warp-per-DAG L-shape executor at every GPU count" ON)
if(GROTOR_BUILD_MULTIGPU)
    if(GROTOR_ENABLE_RTCORE)
        message(FATAL_ERROR "The multi-GPU target requires CPU batching (GROTOR_ENABLE_RTCORE=OFF)")
    endif()
    find_package(MPI REQUIRED COMPONENTS CXX)
    find_path(NVSHMEM_INCLUDE_DIR nvshmem.h HINTS ${NVSHMEM_ROOT} ENV NVSHMEM_HOME PATH_SUFFIXES include)
    find_library(NVSHMEM_HOST_LIBRARY nvshmem_host HINTS ${NVSHMEM_ROOT} ENV NVSHMEM_HOME PATH_SUFFIXES lib lib64)
    if(NOT NVSHMEM_INCLUDE_DIR OR NOT NVSHMEM_HOST_LIBRARY)
        message(FATAL_ERROR "Set NVSHMEM_ROOT to an NVSHMEM 3.x installation")
    endif()
    add_executable(GRotor_multi src/main.cu)
    set_target_properties(GRotor_multi PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17
        CXX_STANDARD_REQUIRED YES CUDA_STANDARD_REQUIRED YES)
    target_include_directories(GRotor_multi PRIVATE ${NVSHMEM_INCLUDE_DIR} ${MPI_CXX_INCLUDE_DIRS})
    target_link_libraries(GRotor_multi PRIVATE grotor_options MPI::MPI_CXX ${NVSHMEM_HOST_LIBRARY})
    target_compile_definitions(GRotor_multi PRIVATE GROTOR_ENABLE_NVSHMEM=1)
    if(GROTOR_MULTI_GRID)
        target_compile_definitions(GRotor_multi PRIVATE GROTOR_MULTI_GRID=1 GROTOR_DENSE_PRESUM=1)
    endif()
    if(GROTOR_MULTI_GRID AND GROTOR_MULTI_LEGACY_LSHAPE)
        find_package(CUDAToolkit REQUIRED)
        find_path(NCCL_INCLUDE_DIR nccl.h HINTS ${NCCL_ROOT} ENV NCCL_HOME "$ENV{HOME}/opt/nccl" PATH_SUFFIXES include)
        find_library(NCCL_LIBRARY nccl HINTS ${NCCL_ROOT} ENV NCCL_HOME "$ENV{HOME}/opt/nccl" PATH_SUFFIXES lib lib64)
        find_library(NVSHMEM_DEVICE_LIBRARY nvshmem_device HINTS ${NVSHMEM_ROOT} ENV NVSHMEM_HOME PATH_SUFFIXES lib lib64)
        if(NOT NCCL_INCLUDE_DIR OR NOT NCCL_LIBRARY OR NOT NVSHMEM_DEVICE_LIBRARY)
            message(FATAL_ERROR "Legacy L-shape requires NCCL and the NVSHMEM device library")
        endif()
        target_include_directories(GRotor_multi PRIVATE ${NCCL_INCLUDE_DIR})
        target_link_libraries(GRotor_multi PRIVATE ${NCCL_LIBRARY} ${NVSHMEM_DEVICE_LIBRARY} CUDA::cuda_driver)
        target_compile_definitions(GRotor_multi PRIVATE GROTOR_MULTI_LEGACY_LSHAPE=1)
        set_target_properties(GRotor_multi PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
    endif()
    option(GROTOR_BUILD_MULTIGPU_TESTS "Build distributed dense-prefix correctness checks" OFF)
    if(GROTOR_BUILD_MULTIGPU_TESTS AND GROTOR_MULTI_GRID)
        add_executable(grotor_multigpu_check script/tests/multigpu_check.cu)

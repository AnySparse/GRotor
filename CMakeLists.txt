cmake_minimum_required(VERSION 3.18)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 89)
endif()
project(GRotor LANGUAGES CXX CUDA)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

add_executable(GRotor src/main.cu)
target_include_directories(GRotor PRIVATE include src third_party)
set_target_properties(GRotor PROPERTIES CXX_STANDARD 17 CUDA_STANDARD 17
                      CXX_STANDARD_REQUIRED YES CUDA_STANDARD_REQUIRED YES)
# Preserve floating-point semantics; keep fast math disabled.
target_compile_options(GRotor PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-UNDEBUG>)
target_compile_definitions(GRotor PRIVATE
    G_ROTOR_RESOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/resources/flute")

if(GROTOR_FULL_COST_REFERENCE)

endif()

if(GROTOR_ENABLE_RTCORE)
    find_package(CUDAToolkit REQUIRED)
    find_path(OPTIX_INCLUDE_DIR optix.h
              HINTS ${OPTIX_ROOT_DIR} ENV OPTIX_ROOT_DIR PATH_SUFFIXES include)
    if(NOT OPTIX_INCLUDE_DIR)
        message(FATAL_ERROR "OptiX headers are required. Set OPTIX_ROOT_DIR to the OptiX SDK directory.")
    endif()
    add_library(grotor_rtcore_ptx OBJECT src/batching/rtcore/color_programs.cu)
    set_target_properties(grotor_rtcore_ptx PROPERTIES CUDA_PTX_COMPILATION ON CUDA_STANDARD 17)
    target_include_directories(grotor_rtcore_ptx PRIVATE ${OPTIX_INCLUDE_DIR})
    target_compile_options(grotor_rtcore_ptx PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
    set(GROTOR_RTCORE_PTX "${CMAKE_CURRENT_BINARY_DIR}/rtcore_batches.ptx")
    add_custom_command(OUTPUT "${GROTOR_RTCORE_PTX}"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_OBJECTS:grotor_rtcore_ptx> "${GROTOR_RTCORE_PTX}"
        DEPENDS grotor_rtcore_ptx $<TARGET_OBJECTS:grotor_rtcore_ptx>
        COMMAND_EXPAND_LISTS)
    add_custom_target(grotor_rtcore_programs DEPENDS "${GROTOR_RTCORE_PTX}")
    add_library(grotor_rtcore_cached_ptx OBJECT src/batching/rtcore/cached_programs.cu)
    set_target_properties(grotor_rtcore_cached_ptx PROPERTIES CUDA_PTX_COMPILATION ON CUDA_STANDARD 17)
    target_include_directories(grotor_rtcore_cached_ptx PRIVATE ${OPTIX_INCLUDE_DIR})
    target_compile_options(grotor_rtcore_cached_ptx PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
    set(GROTOR_RTCORE_CACHED_PTX "${CMAKE_CURRENT_BINARY_DIR}/rtcore_cached.ptx")
    add_custom_command(OUTPUT "${GROTOR_RTCORE_CACHED_PTX}"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_OBJECTS:grotor_rtcore_cached_ptx> "${GROTOR_RTCORE_CACHED_PTX}"
        DEPENDS grotor_rtcore_cached_ptx $<TARGET_OBJECTS:grotor_rtcore_cached_ptx>
        COMMAND_EXPAND_LISTS)
    add_custom_target(grotor_rtcore_cached_programs DEPENDS "${GROTOR_RTCORE_CACHED_PTX}")
    add_dependencies(GRotor grotor_rtcore_cached_programs)
    add_dependencies(GRotor grotor_rtcore_programs)
    target_include_directories(GRotor PRIVATE ${OPTIX_INCLUDE_DIR})
    target_link_libraries(GRotor PRIVATE CUDA::cuda_driver CUDA::cudart ${CMAKE_DL_LIBS})
    target_compile_definitions(GRotor PRIVATE GROTOR_ENABLE_RTCORE=1
        GROTOR_RTCORE_PTX_PATH="${GROTOR_RTCORE_PTX}"
        GROTOR_RTCORE_CACHED_PTX_PATH="${GROTOR_RTCORE_CACHED_PTX}")

option(GROTOR_BUILD_ROUTING_ABLATION "Build cumulative routing ablation executables" OFF)
if(GROTOR_BUILD_ROUTING_ABLATION)
    if(NOT GROTOR_ENABLE_RTCORE)
        message(FATAL_ERROR "Routing ablation requires GROTOR_ENABLE_RTCORE=ON")
    endif()
    foreach(variant IN ITEMS original bit schedule rtcore)
        set(target grotor_ablation_${variant})
        add_executable(${target} src/main.cu)
        set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
        target_include_directories(${target} PRIVATE include src third_party)
        target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-UNDEBUG>)
        target_compile_definitions(${target} PRIVATE GROTOR_REPORT_ABLATION_CONFIG=1
            GROTOR_ABLATION_VARIANT="${variant}"
            G_ROTOR_RESOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/resources/flute")
        if(variant STREQUAL "original")
            target_compile_definitions(${target} PRIVATE GROTOR_FULL_COST_REFERENCE=1 GROTOR_SERIAL_DAG=1)
        elseif(variant STREQUAL "bit")
            target_compile_definitions(${target} PRIVATE GROTOR_SERIAL_DAG=1)
        endif()
        if(variant STREQUAL "rtcore")
            target_include_directories(${target} PRIVATE ${OPTIX_INCLUDE_DIR})
            target_compile_definitions(${target} PRIVATE GROTOR_ENABLE_RTCORE=1
                GROTOR_RTCORE_PTX_PATH="${GROTOR_RTCORE_PTX}"
                GROTOR_RTCORE_CACHED_PTX_PATH="${GROTOR_RTCORE_CACHED_PTX}")
            target_link_libraries(${target} PRIVATE CUDA::cuda_driver CUDA::cudart ${CMAKE_DL_LIBS})
            add_dependencies(${target} grotor_rtcore_programs grotor_rtcore_cached_programs)
        endif()
    endforeach()
    enable_testing()
    add_executable(grotor_global_depth_check script/tests/dag_schedule_check.cu)
    set_target_properties(grotor_global_depth_check PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
    target_include_directories(grotor_global_depth_check PRIVATE include src third_party)
    target_compile_options(grotor_global_depth_check PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;-UNDEBUG>)
    target_compile_definitions(grotor_global_depth_check PRIVATE GROTOR_SERIAL_DAG=1)
    add_test(NAME global_depth_scheduler COMMAND grotor_global_depth_check)
    set_tests_properties(global_depth_scheduler PROPERTIES RUN_SERIAL TRUE TIMEOUT 180)
endif()
